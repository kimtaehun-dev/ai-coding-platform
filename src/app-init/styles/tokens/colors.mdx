export function SectionHeader({ eyebrow, title, description }) {
  return (
    <header style={{ marginBottom: 16 }}>
      {eyebrow ? (
        <div style={{ fontSize: 11, fontWeight: 600, color: 'var(--color-foreground-subtle)', textTransform: 'uppercase', letterSpacing: 0.6, marginBottom: 4 }}>
          {eyebrow}
        </div>
      ) : null}
      <h2 style={{ fontSize: 22, fontWeight: 600, color: 'var(--color-foreground)', margin: '0 0 6px 0' }}>{title}</h2>
      <p style={{ fontSize: 13, color: 'var(--color-foreground-muted)', margin: 0, lineHeight: 1.55 }}>{description}</p>
    </header>
  )
}

export function Divider() {
  return <hr style={{ border: 0, borderTop: '1px solid var(--color-border-muted)', margin: '40px 0' }} />
}

export function Swatch({ cssVar, label, sub }) {
  return (
    <div style={{ display: 'flex', alignItems: 'center', gap: 12, minWidth: 0 }}>
      <div style={{ width: 44, height: 44, borderRadius: 8, background: `var(${cssVar})`, border: '1px solid var(--color-border)', flexShrink: 0 }} aria-hidden />
      <div style={{ display: 'flex', flexDirection: 'column', minWidth: 0 }}>
        <code style={{ fontSize: 12, fontFamily: 'var(--font-mono)', color: 'var(--color-foreground)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{label}</code>
        {sub ? (
          <span style={{ fontSize: 11, color: 'var(--color-foreground-muted)', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{sub}</span>
        ) : null}
      </div>
    </div>
  )
}

export function ThemePanel({ scheme, children }) {
  const isDark = scheme === 'dark'
  return (
    <div className={isDark ? 'dark' : undefined} style={{ background: 'var(--color-background)', color: 'var(--color-foreground)', padding: 20, borderRadius: 12, border: '1px solid var(--color-border)' }}>
      <div style={{ fontSize: 12, fontWeight: 600, color: 'var(--color-foreground-muted)', marginBottom: 16, textTransform: 'uppercase', letterSpacing: 0.4 }}>
        {isDark ? 'Dark mode' : 'Light mode'}
      </div>
      {children}
    </div>
  )
}

export function SemanticGroup({ group }) {
  return (
    <section style={{ marginBottom: 24 }}>
      <h4 style={{ fontSize: 14, fontWeight: 600, color: 'var(--color-foreground)', margin: '0 0 4px 0' }}>{group.title}</h4>
      <p style={{ fontSize: 12, color: 'var(--color-foreground-muted)', margin: '0 0 12px 0' }}>{group.description}</p>
      <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fill, minmax(220px, 1fr))', gap: 12 }}>
        {group.tokens.map((t) => (
          <Swatch key={t.name} cssVar={`--color-${t.name}`} label={`--color-${t.name}`} sub={t.usage} />
        ))}
      </div>
    </section>
  )
}
