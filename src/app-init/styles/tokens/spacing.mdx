export function SectionHeader({ eyebrow, title, description }) {
  return (
    <header style={{ marginBottom: 20 }}>
      {eyebrow ? (
        <div style={{ fontSize: 11, fontWeight: 600, color: 'var(--color-foreground-subtle)', textTransform: 'uppercase', letterSpacing: 0.6, marginBottom: 4 }}>
          {eyebrow}
        </div>
      ) : null}
      <h2 style={{ fontSize: 22, fontWeight: 600, color: 'var(--color-foreground)', margin: '0 0 6px 0' }}>{title}</h2>
      <p style={{ fontSize: 13, color: 'var(--color-foreground-muted)', margin: 0, lineHeight: 1.55 }}>{description}</p>
    </header>
  )
}

export function Divider() {
  return <hr style={{ border: 0, borderTop: '1px solid var(--color-border-muted)', margin: '40px 0' }} />
}

export function SpacingBar({ token }) {
  return (
    <div style={{ display: 'grid', gridTemplateColumns: '120px 100px 1fr 1fr', gap: 16, alignItems: 'center', padding: '10px 0', borderBottom: '1px solid var(--color-border-muted)' }}>
      <code style={{ fontSize: 12, fontFamily: 'var(--font-mono)', color: 'var(--color-foreground)' }}>spacing-{token.key}</code>
      <span style={{ fontSize: 12, color: 'var(--color-foreground-muted)' }}>{token.px} / {token.rem}</span>
      <div style={{ height: 16, width: `var(--spacing-${token.key})`, background: 'var(--color-primary)', borderRadius: 2 }} aria-label={`${token.px} 너비 시각화`} />
      <span style={{ fontSize: 12, color: 'var(--color-foreground-muted)' }}>{token.usage}</span>
    </div>
  )
}
