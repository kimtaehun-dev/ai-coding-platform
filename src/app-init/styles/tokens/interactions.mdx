export function StateSwatch({ token, label, sub, onToken }) {
  const bg = `var(--color-${token})`
  const fg = onToken ? `var(--color-${onToken})` : 'var(--color-foreground)'
  return (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 8 }}>
      <div style={{ background: bg, color: fg, padding: '14px 16px', borderRadius: 8, display: 'flex', alignItems: 'center', justifyContent: 'space-between', minHeight: 48, border: '1px solid var(--color-border-muted)' }}>
        <span style={{ fontSize: 13, fontWeight: 500 }}>{label}</span>

      </div>
      <code style={{ fontSize: 11, fontFamily: 'var(--font-mono)', color: 'var(--color-foreground-muted)' }}>--color-{token}</code>
    </div>
  )
}

export function LiveButton({ variant, label }) {
  const variantStyles = {
    primary: { bg: 'var(--color-primary)', bgHover: 'var(--color-primary-hover)', bgActive: 'var(--color-primary-active)', fg: 'var(--color-on-primary)' },
    secondary: { bg: 'var(--color-secondary)', bgHover: 'var(--color-secondary-hover)', bgActive: 'var(--color-secondary-active)', fg: 'var(--color-on-secondary)' },
    error: { bg: 'var(--color-error)', bgHover: 'var(--color-error-hover)', bgActive: 'var(--color-error-active)', fg: 'var(--color-on-error)' },
  }
  const v = variantStyles[variant]
  return (
    <button
      style={{
        background: v.bg,
        color: v.fg,
        border: 'none',
        padding: '10px 18px',
        borderRadius: 'var(--radius-md)',
        fontSize: 14,
        fontWeight: 500,
        cursor: 'pointer',
        transition: 'background 120ms ease',
        outline: 'none',
      }}
      onMouseEnter={(e) => { e.currentTarget.style.background = v.bgHover }}
      onMouseLeave={(e) => { e.currentTarget.style.background = v.bg }}
      onMouseDown={(e) => { e.currentTarget.style.background = v.bgActive }}
      onMouseUp={(e) => { e.currentTarget.style.background = v.bgHover }}
      onFocus={(e) => { e.currentTarget.style.boxShadow = `0 0 0 3px var(--color-focus-ring)` }}
      onBlur={(e) => { e.currentTarget.style.boxShadow = 'none' }}
    >
      {label}
    </button>
  )
}
