export const SAMPLE_KO = '디자인 시스템에서 일관성은 신뢰의 기반이다.'
export const SAMPLE_EN = 'The quick brown fox jumps over the lazy dog.'
export const SAMPLE_CODE = 'const tokens = readCssVars();'

export function SectionHeader({ eyebrow, title, description }) {
  return (
    <header style={{ marginBottom: 20 }}>
      {eyebrow ? (
        <div style={{ fontSize: 11, fontWeight: 600, color: 'var(--color-foreground-subtle)', textTransform: 'uppercase', letterSpacing: 0.6, marginBottom: 4 }}>
          {eyebrow}
        </div>
      ) : null}
      <h2 style={{ fontSize: 22, fontWeight: 600, color: 'var(--color-foreground)', margin: '0 0 6px 0' }}>{title}</h2>
      <p style={{ fontSize: 13, color: 'var(--color-foreground-muted)', margin: 0, lineHeight: 1.55 }}>{description}</p>
    </header>
  )
}

export function Divider() {
  return <hr style={{ border: 0, borderTop: '1px solid var(--color-border-muted)', margin: '40px 0' }} />
}

export function ScaleRow({ scale, group }) {
  const isCode = group === 'Utility' && scale.token.startsWith('code')
  const sample = isCode ? SAMPLE_CODE : `${SAMPLE_KO} ${SAMPLE_EN}`
  const sp = scale.specs || {}
  return (
    <div style={{ display: 'grid', gridTemplateColumns: '170px 1fr 150px', gap: 20, alignItems: 'baseline', padding: '16px 0', borderBottom: '1px solid var(--color-border-muted)' }}>
      <div>
        <code style={{ fontSize: 12, fontFamily: 'var(--font-mono)', color: 'var(--color-foreground)', display: 'block' }}>text-{scale.token}</code>
        <span style={{ fontSize: 11, color: 'var(--color-foreground-muted)', display: 'block', marginTop: 2 }}>{scale.usage}</span>
      </div>
      <div className={`text-${scale.token}`} style={{ color: 'var(--color-foreground)', fontFamily: isCode ? 'var(--font-mono)' : undefined, minWidth: 0, overflow: 'hidden', textOverflow: 'ellipsis' }}>
        {sample}
      </div>
      <code style={{ fontSize: 11, fontFamily: 'var(--font-mono)', color: 'var(--color-foreground-subtle)', lineHeight: 1.6, whiteSpace: 'nowrap' }}>
        {sp.size} · lh {sp.lh}
        <br />
        w {sp.weight} · ls {sp.ls}
        {sp.family === 'mono' ? <><br />mono</> : null}
      </code>
    </div>
  )
}
