export function SectionHeader({ eyebrow, title, description }) {
  return (
    <header style={{ marginBottom: 20 }}>
      {eyebrow ? (
        <div style={{ fontSize: 11, fontWeight: 600, color: 'var(--color-foreground-subtle)', textTransform: 'uppercase', letterSpacing: 0.6, marginBottom: 4 }}>
          {eyebrow}
        </div>
      ) : null}
      <h2 style={{ fontSize: 22, fontWeight: 600, color: 'var(--color-foreground)', margin: '0 0 6px 0' }}>{title}</h2>
      <p style={{ fontSize: 13, color: 'var(--color-foreground-muted)', margin: 0, lineHeight: 1.55 }}>{description}</p>
    </header>
  )
}

export function Divider() {
  return <hr style={{ border: 0, borderTop: '1px solid var(--color-border-muted)', margin: '40px 0' }} />
}

export function ShadowCard({ level, usage, scheme }) {
  return (
    <div className={scheme === 'dark' ? 'dark' : undefined} style={{ background: 'var(--color-background)', padding: 24, borderRadius: 12 }}>
      <div style={{ background: 'var(--color-surface-elevated)', padding: 24, borderRadius: 12, boxShadow: `var(--shadow-level-${level})`, color: 'var(--color-foreground)', minHeight: 100, display: 'flex', flexDirection: 'column', justifyContent: 'center', gap: 4 }}>
        <code style={{ fontSize: 12, fontFamily: 'var(--font-mono)', color: 'var(--color-foreground)' }}>shadow-level-{level}</code>
        <span style={{ fontSize: 11, color: 'var(--color-foreground-muted)' }}>{usage}</span>
      </div>
    </div>
  )
}

export function SchemeGrid({ scheme }) {
  return (
    <div className={scheme === 'dark' ? 'dark' : undefined} style={{ padding: 24, background: 'var(--color-surface)', borderRadius: 12, border: '1px solid var(--color-border)' }}>
      <div style={{ fontSize: 12, fontWeight: 600, color: 'var(--color-foreground-muted)', marginBottom: 16, textTransform: 'uppercase', letterSpacing: 0.4 }}>
        {scheme === 'dark' ? 'Dark mode' : 'Light mode'}
      </div>
      <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fill, minmax(200px, 1fr))', gap: 16 }}>
        {ELEVATION_TOKENS.map((t) => (
          <ShadowCard key={t.level} level={t.level} usage={t.usage} scheme={scheme} />
        ))}
      </div>
    </div>
  )
}
